"lazy" true) */}} {{- $img := resources.Get .src -}} {{- if not $img -}} {{- warnf "Afbeelding %q niet gevonden in assets/" .src -}} {{- else -}} {{- $breedtes := .breedtes | default (slice 480 960 1600) -}} {{- $srcset := slice -}} {{- $fallback := $img -}} {{- range $breedtes -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- $fallback = $r -}} {{- end -}} {{- end -}} {{- if not $srcset -}} {{- $fallback = $img.Process "webp" -}} {{- $srcset = slice (printf "%s %dw" $fallback.RelPermalink $fallback.Width) -}} {{- end -}} {{- end -}}
